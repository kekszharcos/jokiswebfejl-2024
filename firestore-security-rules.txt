rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ===== USERS COLLECTION =====
    match /Users/{userId} {
      // Allow authenticated users to read any user document (for friend search, etc.)
      allow read: if request.auth != null;
      
      // Users can write to their own profile
      allow write: if request.auth != null && request.auth.uid == userId;
      
      // CRITICAL: Allow user creation during signup/Google login flow
      allow create: if request.auth != null && request.auth.uid == userId;
      
      // Allow users to update their own profile
      allow update: if request.auth != null && request.auth.uid == userId;
      
      // Allow users to delete their own profile
      allow delete: if request.auth != null && request.auth.uid == userId;
    }
    
    // ===== MESSAGES COLLECTION =====
    match /Messages/{messageId} {
      // Users can read messages where they are the owner
      allow read: if request.auth != null && 
                     request.auth.uid == resource.data.ownerId;
      
      // Users can create messages where they are the owner
      allow create: if request.auth != null && 
                       request.auth.uid == request.resource.data.ownerId;
      
      // Users can update/delete their own messages
      allow update, delete: if request.auth != null && 
                               request.auth.uid == resource.data.ownerId;
      
      // Allow reading messages if user is part of the chat
      allow read: if request.auth != null && 
                     isUserInChat(request.auth.uid, resource.data.chatId);
    }
    
    // ===== CHATS COLLECTION =====
    match /Chats/{chatId} {
      // Users can read/write chats where they are uid1 or uid2
      allow read, write: if request.auth != null && 
                            (request.auth.uid == resource.data.uid1 || 
                             request.auth.uid == resource.data.uid2);
      
      // Users can create chats where they are uid1 or uid2
      allow create: if request.auth != null && 
                       (request.auth.uid == request.resource.data.uid1 || 
                        request.auth.uid == request.resource.data.uid2);
    }
    
    // ===== GROUPS COLLECTION =====
    match /Groups/{groupId} {
      // Users can read groups they are members of
      allow read: if request.auth != null && 
                     request.auth.uid in resource.data.users;
      
      // Only group owner can write/update group
      allow write: if request.auth != null && 
                      request.auth.uid == resource.data.owner;
      
      // Users can create groups where they are the owner
      allow create: if request.auth != null && 
                       request.auth.uid == request.resource.data.owner;
    }
    
    // ===== HELPER FUNCTIONS =====
    function isUserInChat(userId, chatId) {
      return exists(/databases/$(database)/documents/Chats/$(chatId)) &&
             (get(/databases/$(database)/documents/Chats/$(chatId)).data.uid1 == userId ||
              get(/databases/$(database)/documents/Chats/$(chatId)).data.uid2 == userId);
    }
    
    // ===== FALLBACK RULE =====
    // Secure fallback - deny all other access
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
