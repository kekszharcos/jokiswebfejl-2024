rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ===== BASIC AUTH REQUIREMENT =====
    // All operations require authentication
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
    
    // ===== USERS COLLECTION - More Restrictive =====
    match /Users/{userId} {
      // Users can only read/write their own profile
      allow read, write: if request.auth != null && request.auth.uid == userId;
      
      // Allow reading other users for friend search (but limit fields if needed)
      allow read: if request.auth != null;
    }
    
    // ===== MESSAGES COLLECTION =====
    match /Messages/{messageId} {
      // Users can only read/write messages they own
      allow read, write: if request.auth != null && 
                            request.auth.uid == resource.data.ownerId;
      
      // Allow creating messages where user is the owner
      allow create: if request.auth != null && 
                       request.auth.uid == request.resource.data.ownerId;
    }
  }
}
